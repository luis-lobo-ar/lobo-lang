  let original = Pair { left: 2, right: 3 };
  let updated = Pair { ...original, left: 8 };
  updated.left + updated.right
}

fn counted() -> I64 {
  fold(index in 4, total: I64 = 0) { total + index }
}

fn stopped() -> I64 {
  while(state: I64 = 0; state < 4) { state + 1 }
}

fn list_order() -> I64 {
  let values = List::prepend(2, List::prepend(9, List::empty<I64>()));
  List::head(values)
}

fn propagate() -> I64 {
  match bump(Ok(8)) { Ok(value) => value, Err(error) => -1 }
}

fn propagate_error() -> Bool {
  match bump(Err("example")) { Ok(value) => false, Err(error) => error == "example" }
}
